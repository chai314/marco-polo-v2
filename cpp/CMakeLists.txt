cmake_minimum_required(VERSION 3.16)
project(vropt LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(Threads REQUIRED)

add_library(vropt INTERFACE)
target_include_directories(vropt INTERFACE include)
target_link_libraries(vropt INTERFACE Threads::Threads)
if(NOT MSVC)
  target_compile_options(vropt INTERFACE -O3 -ffp-contract=off -Wall -Wextra)
endif()
if(MINGW)
  target_link_options(vropt INTERFACE -static)
endif()

add_library(vropt_core SHARED src/capi.cpp)
target_link_libraries(vropt_core PRIVATE vropt)
add_executable(vropt_cli apps/vropt_cli.cpp)
target_link_libraries(vropt_cli PRIVATE vropt)

enable_testing()
add_executable(vropt_tests tests/test_vropt.cpp)
target_link_libraries(vropt_tests PRIVATE vropt)
add_test(NAME vropt_tests COMMAND vropt_tests)
